#
# These are separated from requirements.txt because they need:
#   1. CUDA_HOME set (point to wherever nvcc lives, e.g. $CONDA_PREFIX)
#   2. nvcc on PATH
#   3. A CUDA-enabled PyTorch already installed
#   4. --no-build-isolation so builds link against the installed torch
#
# Manual install (after the environment that provides Python is active):
#   conda install -y -c nvidia cuda-toolkit    # provides nvcc in $CONDA_PREFIX
#   export CUDA_HOME=$CONDA_PREFIX
#   pip install -r ml/requirements-cuda.txt --no-build-isolation
